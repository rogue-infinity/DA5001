\section{Experimental Setup}
%% ============================================================

\subsection*{Model \& Data}
Target model: \texttt{dllm-hub/Qwen3-0.6B-diffusion-mdlm-v0.1} (MDLM, 0.6B parameters).
Membership split: \texttt{ngram\_13\_0.8} (target), \texttt{ngram\_13\_0.2} (shadow surrogate).
Domains evaluated: arxiv, github, hackernews, pubmed\_central, wikipedia, pile\_cc (all 6 MIMIR domains).

\subsection*{Shadow Model Protocol}
\begin{itemize}
  \item $K=3$ shadow models fine-tuned on pooled surrogate data (all 6 MIMIR domains pooled, $\approx$5{,}695 members total, 1{,}000 per shadow model after chunking).
  \item Pooled cross-domain shadow prevents data starvation for sparse domains (e.g.\ hackernews had only 35 domain-specific surrogates).
  \item 46-dimensional feature vector extracted at 4 diffusion timesteps ($t \in \{0.05, 0.20, 0.35, 0.50\}$): ELBO, predicted entropy, mask consistency, attention entropy, gradient norms.
  \item XGBoost trained on shadow features, transferred to target domain without labels.
\end{itemize}

\subsection*{Conditions}
\begin{center}
\begin{tabular}{lll}
\toprule
\textbf{Label} & \textbf{Description} & \textbf{Dims} \\
\midrule
A: Oracle   & 5-fold CV on target (white-box upper bound, replays Run3) & 46 \\
B: Shadow   & Shadow $\to$ target transfer, full feature set             & 46 \\
C: ELBO+H   & Transfer, permutation-invariant ELBO + entropy only        & 8  \\
D: Attn     & Transfer, attention dims only (negative control)           & 16 \\
E: Pruned   & Transfer, drop attention dims (ablation of B)              & 30 \\
\bottomrule
\end{tabular}
\end{center}

\subsection*{Baselines}
\begin{itemize}
  \item \textbf{SAMA} (Subset-Aggregated MIA): $T=4$ steps, $N=128$ subsets, $m=10$ tokens, $\alpha \in [5\%, 50\%]$. Run in Run3; scores loaded directly.
  \item \textbf{Oracle} (Condition A): white-box upper bound; requires full model access and target labels.
\end{itemize}

\begin{table}[ht]

\noindent\textbf{Key findings (5-domain preliminary):}
\begin{enumerate}
  \item Shadow transfer (B/E) closes to within $\Delta=0.020$--$0.025$ AUC of the white-box oracle, despite having no target labels.
  \item Shadow-B and Shadow-E are nearly identical ($\Delta=0.005$ AUC), meaning the 16 attention dimensions carry no useful cross-domain signal.
  \item SAMA trails shadow transfer by $\approx$0.036 AUC on average, with a much larger gap at low FPR (TPR@1\%: 0.053 vs 0.213).
  \item ELBO features at $t \in \{0.20, 0.35, 0.50\}$ are the primary signal carriers (Hellinger $\approx$0.40--0.42).
  \item Condition D (attention-only) collapses to AUC $\approx$0.53, validating the feature pruning rationale.
\end{enumerate}
